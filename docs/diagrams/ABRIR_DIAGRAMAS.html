12. Entity-Relationship Diagram (ERD)

erDiagram
  USERS ||--o{ TICKETS : requests_or_assigned
  TICKETS ||--o{ TICKET_NOTES : contains
  TICKETS ||--o{ ATTACHMENTS : has_files
  TICKETS |o--o| AI_SESSIONS : escalated_from
  KB_ARTICLES ||--|{ KB_VERSIONS : versions
  USERS ||--o{ AUDIT_LOGS : triggers
  USERS { UUID user_id PK VARCHAR email UK VARCHAR password_hash ENUM role }
  TICKETS { SERIAL ticket_id PK UUID requester_id FK UUID assigned_tech_id FK ENUM category ENUM priority ENUM status ENUM resolved_by BOOLEAN sla_breached TIMESTAMPTZ created_at }
  TICKET_NOTES { UUID note_id PK INTEGER ticket_id FK TEXT content BOOLEAN is_internal }
  ATTACHMENTS { UUID attachment_id PK INTEGER ticket_id FK VARCHAR mime_type INTEGER file_size_bytes }
  KB_ARTICLES { UUID article_id PK VARCHAR title ENUM category ENUM status BOOLEAN is_vectorized }
  AI_SESSIONS { UUID session_id PK INTEGER ticket_id FK INTEGER failed_attempts JSONB transcript_json }
  AUDIT_LOGS { BIGSERIAL log_id PK UUID actor_id FK TIMESTAMPTZ timestamp VARCHAR action JSONB old_value JSONB new_value }
